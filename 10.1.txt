import pandas as pd

m = pd.read_parquet("/Volumes/bootcamp/northern_star/raw_2025_26/matches_2025_26.parquet")

1.

# 1. Oba tima dala bar po gol
oba_gol = m[(m['home_goals'] > 0) & (m['away_goals'] > 0)]
print(oba_gol)

# 2. Domaćin pobedio sa tačno jednim golom razlike
jedan_razlike = m[(m['home_goals'] - m['away_goals']) == 1]
print(jedan_razlike)

# 3. Broj utakmica završenih 0:0
bez_golova = m[(m['home_goals'] == 0) & (m['away_goals'] == 0)]
print("Broj utakmica 0:0 =", len(bez_golova))

# 4. Top 10 utakmica sa najviše golova
m['ukupno_golova'] = m['home_goals'] + m['away_goals']

top10 = m.sort_values(
    'ukupno_golova',
    ascending=False
).head(10)

print(top10)

2.

tim = "BMC"

# 1. Sve utakmice tima
utakmice = m[
    (m['home_team'] == tim) |
    (m['away_team'] == tim)
]

print(utakmice)

# 2. Pobede, porazi, nerešeno

pobede = len(
    m[
        ((m['home_team'] == tim) & (m['home_goals'] > m['away_goals'])) |
        ((m['away_team'] == tim) & (m['away_goals'] > m['home_goals']))
    ]
)

porazi = len(
    m[
        ((m['home_team'] == tim) & (m['home_goals'] < m['away_goals'])) |
        ((m['away_team'] == tim) & (m['away_goals'] < m['home_goals']))
    ]
)

nereseno = len(
    m[
        ((m['home_team'] == tim) |
         (m['away_team'] == tim))
        &
        (m['home_goals'] == m['away_goals'])
    ]
)

print("Pobede:", pobede)
print("Porazi:", porazi)
print("Nereseno:", nereseno)

# 3. Golovi kod kuće i u gostima

golovi_kuca = m[
    m['home_team'] == tim
]['home_goals'].sum()

golovi_gosti = m[
    m['away_team'] == tim
]['away_goals'].sum()

print("Kod kuce:", golovi_kuca)
print("U gostima:", golovi_gosti)

# 4. Utakmica sa najviše postignutih golova

utakmice = utakmice.copy()

utakmice['golovi_tima'] = 0

utakmice.loc[
    utakmice['home_team'] == tim,
    'golovi_tima'
] = utakmice['home_goals']

utakmice.loc[
    utakmice['away_team'] == tim,
    'golovi_tima'
] = utakmice['away_goals']

najvise = utakmice.sort_values(
    'golovi_tima',
    ascending=False
).head(1)

print(najvise)

# 1. Prikazati sve različite stadione na kojima su odigrane utakmice

print("1. Različiti stadioni:")
print(m["stadium"].unique())


# 2. Pronaći stadion na kojem je odigrano najviše utakmica

print("\n2. Stadion sa najviše utakmica:")
print(m["stadium"].value_counts().head(1))


# 3. Za svaki stadion izračunati prosečnu posećenost

print("\n3. Prosečna posećenost po stadionu:")

prosecna_posecenost = (
    m.groupby("stadium")["attendance"]
    .mean()
)

print(prosecna_posecenost)


# 4. Prikazati pet stadiona sa najvećom prosečnom posećenošću

print("\n4. Top 5 stadiona po prosečnoj posećenosti:")

top5_stadiona = (
    m.groupby("stadium")["attendance"]
    .mean()
    .sort_values(ascending=False)
    .head(5)
)

print(top5_stadiona)
