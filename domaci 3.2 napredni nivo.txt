--1. Pronađite pet igrača sa najvećim brojem asistencija u sezoni 2024/25. Prikažite ime igrača i broj asistencija.
SELECT
    p.first_name,
    p.last_name,
    COUNT(*) AS assists
FROM match_events me
JOIN players p
    ON me.related_player_id = p.player_id
JOIN matches m
    ON me.match_id = m.match_id
WHERE m.season = '2024/25'
  AND me.event_type IN ('goal', 'penalty_goal')
GROUP BY p.player_id, p.first_name, p.last_name
ORDER BY assists DESC
LIMIT 5;

--2. Pronađite tri igrača sa najvećim ukupnim brojem odigranih minuta u sezoni 2024/25. Prikažite ime igrača, klub i ukupan broj minuta.
SELECT
p.first_name,
p.last_name,
t.team_name,
SUM(s.minutes_played) AS total_minutes
FROM player_match_stats s
JOIN players p
ON s.player_id = p.player_id
JOIN teams t
ON s.team_id = t.team_id
JOIN matches m
ON s.match_id = m.match_id
WHERE m.season = '2024/25'
GROUP BY p.player_id, p.first_name, p.last_name, t.team_name
ORDER BY total_minutes DESC
LIMIT 3;

--3. Za svaku sezonu prikažite broj isključenja po klubovima. Računajte događaje red_card i second_yellow. Uključite samo klubove koji su imali najmanje jedno isključenje.
SELECT
    m.season,
    t.team_name,
    COUNT(*) AS red_cards
FROM match_events me
JOIN matches m
    ON me.match_id = m.match_id
JOIN teams t
    ON me.team_id = t.team_id
WHERE me.event_type IN ('red_card', 'second_yellow')
GROUP BY m.season, t.team_name
HAVING COUNT(*) > 0
ORDER BY m.season, red_cards DESC;
