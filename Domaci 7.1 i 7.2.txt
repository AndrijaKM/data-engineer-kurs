import math

lista = [1.5, 2.5, 3.5, 4.5]

s = sum(lista) / len(lista)

zbir = 0

for x in lista:
    zbir = zbir + (x - s) ** 2

d = math.sqrt(zbir / (len(lista) - 1))

print("Srednja vrednost:", s)
print("Standardna devijacija:", d)
